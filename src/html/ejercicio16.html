const sumarRango = (inicial, final) => { let suma = 0; for (let i = inicial; i <= final; i++) { suma += i; } return suma; }; const numeroInicial = parseInt(prompt("Ingrese el número inicial del rango:")); const numeroFinal = parseInt(prompt("Ingrese el número final del rango:")); if (numeroInicial <= numeroFinal) { const resultado = sumarRango(numeroInicial, numeroFinal); document.write(`La suma de todos los números en el rango de ${numeroInicial} a ${numeroFinal} es: ${resultado}`); } else { document.write(`Error: El número inicial (${numeroInicial}) debe ser menor o igual que el número final (${numeroFinal}). Recarga la página e intenta de nuevo.`); }
